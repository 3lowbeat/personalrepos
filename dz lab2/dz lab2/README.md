# Домашнее задание к работе 2

## Условие задачи
В первом сосуде содержится V1 литров воды температуры t1, во втором – V2 литров
воды температуры t2, в третьем – V3 литров воды температуры t3. Воду слили в один
сосуд. Найти объем и температуру воды в этом сосуде.

## 1. Алгоритм и блок-схема
1. **Начало**
2. Задание конкретных значений переменных
- V1=5 л
- V2=2 л
- V3=4 л
- t1=20 
- t2=5
- t3=50
3. Вычисление полученного объема воды
V = V1 + V2 + V3 л
4. Вычисление полученной температуры воды
t = (V1 * t1 + V2 * t2 + V3 * t3) / V 
4. Вывод результатов подсчетов с подстановкой значений в текст
5. **Конец**

## 2. Реализация программы 
#include <stdio.h>
#include <locale.h>


void main() 
{
	//Шаг 1. Задание конкретных значений переменных
	int V1=5;
	int V2=2;
	int V3=4;

	int t1=20;
	int t2=5;
	int t3=50;

	//Шаг 2. Вычисление полученного объема воды 
	int V = V1 + V2 + V3;

	//Шаг 3. Вычисление полученой температуры воды
	float t = (V1 * t1 + V2 * t2 + V3 * t3) / V;

	//Шаг 4. Форматированный вывод результатов
	setlocale(LC_CTYPE, "RUS");
	printf("ВЫЧИСЛЕНИЕ ОБЪЕМА И ТЕМПЕРАТУРЫ ВОДЫ В СОСУДЕ\n");
	printf("=============================================\n");
	printf("В первом сосуде содержится %d литров воды температуры %d°\nВо втором – %d литров воды температуры %d°\nВ третьем – %d литров воды температуры %d°.\n", V1, t1, V2, t2, V3, t3);
	printf("Воду слили в один сосуд.\n");
	printf("=============================================\n");
	printf("РАСЧЁТ\n");
	printf("Общий объём воды в сосуде: %d л + %d л + %d л = %d л\n", V1, V2, V3, V);
	printf("Температура воды в сосуде: (%d л * %d° + %d л * %d° + %d л * %d°)//%d = %.2f°\n", V1, t1, V2, t2, V3, t3, V, t);
	return 0;

}
